# 매매 대시보드

## 1. 미청산 포지션
```dataview
TABLE WITHOUT ID file.link AS 매매, direction AS 방향, entry_date AS 진입일, entry_price AS 진입가, stop_price AS 손절가, target_price AS 목표가, qty AS 수량
FROM "01_Trades"
WHERE type = "trade" AND status = "open"
SORT entry_date ASC
```

## 2. 전체 성과 요약
```dataviewjs
const t = dv.pages('"01_Trades"').where(p => p.type === "trade" && p.status === "closed");
const pnl = t.pnl.array().map(x => x || 0);
const wins = pnl.filter(x => x > 0), losses = pnl.filter(x => x < 0);
const n = pnl.length;
const sum = a => a.reduce((x, y) => x + y, 0);
const avgW = wins.length ? sum(wins) / wins.length : 0;
const avgL = losses.length ? Math.abs(sum(losses) / losses.length) : 0;
const winRate = n ? wins.length / n : 0;
const avgR = n ? sum(t.r_multiple.array().map(x => x || 0)) / n : 0;
const ab = t.where(p => p.grade === "A" || p.grade === "B").length;
dv.table(["청산 매매", "승률", "총 손익", "손익비", "기대값(1회)", "평균 R", "계획 준수율"], [[
  n,
  (winRate * 100).toFixed(1) + "%",
  sum(pnl).toLocaleString(),
  avgL ? (avgW / avgL).toFixed(2) : "-",
  Math.round(winRate * avgW - (1 - winRate) * avgL).toLocaleString(),
  avgR.toFixed(2),
  n ? (ab / n * 100).toFixed(1) + "%" : "-"
]]);
```

## 3. 월별 성과
```dataviewjs
const t = dv.pages('"01_Trades"').where(p => p.type === "trade" && p.status === "closed" && p.exit_date);
const g = t.groupBy(p => dv.date(p.exit_date).toFormat("yyyy-MM"));
dv.table(["월", "매매 수", "승률", "손익", "평균 R"],
  g.sort(x => x.key, "desc").map(x => {
    const r = x.rows, n = r.length;
    const w = r.where(p => p.pnl > 0).length;
    const s = r.pnl.array().reduce((a, b) => a + (b || 0), 0);
    const ar = r.r_multiple.array().reduce((a, b) => a + (b || 0), 0) / n;
    return [x.key, n, (w / n * 100).toFixed(1) + "%", s.toLocaleString(), ar.toFixed(2)];
  }));
```

## 4. 전략별 성과
```dataviewjs
const t = dv.pages('"01_Trades"').where(p => p.type === "trade" && p.status === "closed");
const g = t.groupBy(p => p.strategy ? p.strategy.path : "미지정");
dv.table(["전략", "매매 수", "승률", "손익", "평균 R"],
  g.map(x => {
    const r = x.rows, n = r.length;
    const w = r.where(p => p.pnl > 0).length;
    const s = r.pnl.array().reduce((a, b) => a + (b || 0), 0);
    const ar = r.r_multiple.array().reduce((a, b) => a + (b || 0), 0) / n;
    return [x.key === "미지정" ? x.key : dv.fileLink(x.key), n, (w / n * 100).toFixed(1) + "%", s.toLocaleString(), ar.toFixed(2)];
  }).sort(x => x[4], "desc"));
```

## 5. 실수 유형별 비용
```dataviewjs
const t = dv.pages('"01_Trades"').where(p => p.type === "trade" && p.status === "closed" && p.mistakes && p.mistakes.length);
const m = {};
for (const p of t) for (const k of p.mistakes) {
  m[k] ??= { n: 0, pnl: 0 };
  m[k].n++; m[k].pnl += p.pnl || 0;
}
dv.table(["실수 유형", "발생 횟수", "관련 손익 합계"],
  Object.entries(m).sort((a, b) => a[1].pnl - b[1].pnl).map(([k, v]) => [k, v.n, v.pnl.toLocaleString()]));
```

## 6. 누적 손익 추이
```dataviewjs
const t = dv.pages('"01_Trades"').where(p => p.type === "trade" && p.status === "closed" && p.exit_date).sort(p => p.exit_date);
let acc = 0;
dv.table(["청산일", "매매", "손익", "누적 손익"],
  t.map(p => { acc += p.pnl || 0; return [dv.date(p.exit_date).toFormat("yyyy-MM-dd"), p.file.link, (p.pnl || 0).toLocaleString(), acc.toLocaleString()]; }));
```

## 7. 최근 매매 10건
```dataview
TABLE WITHOUT ID file.link AS 매매, strategy AS 전략, pnl AS 손익, r_multiple AS R, grade AS 등급, mistakes AS 실수
FROM "01_Trades"
WHERE type = "trade" AND status = "closed"
SORT exit_date DESC
LIMIT 10
```
