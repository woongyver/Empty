## 1. 주간 성과
```dataviewjs
const cur = dv.current();
const s = dv.date(cur.start), e = dv.date(cur.end);
const t = dv.pages('"01_Trades"').where(p => p.type === "trade" && p.status === "closed" && p.exit_date && dv.date(p.exit_date) >= s && dv.date(p.exit_date) <= e);
const n = t.length;
const wins = t.where(p => p.pnl > 0).length;
const sum = t.pnl.array().reduce((a, b) => a + (b || 0), 0);
const avgR = n ? t.r_multiple.array().reduce((a, b) => a + (b || 0), 0) / n : 0;
dv.table(["매매 수", "승률", "총 손익", "평균 R"],
  [[n, n ? (wins / n * 100).toFixed(1) + "%" : "-", sum.toLocaleString(), avgR.toFixed(2)]]);
dv.table(["매매", "전략", "손익", "R", "등급", "실수"],
  t.sort(p => p.exit_date).map(p => [p.file.link, p.strategy, p.pnl, p.r_multiple, p.grade, p.mistakes]));
```
